# rekordbox-toolkit Python dependencies
#
# Install with: pip install -r requirements.txt
#
# System binaries required separately (see README):
#   macOS:   brew install ffmpeg chromaprint
#   Windows: winget install ffmpeg  (fpcalc: https://acoustid.org/chromaprint)
#
# pyrekordbox 0.4.4 is the tested version. The add_content, add_to_playlist,
# and update_content_path call signatures are version-sensitive — test after
# any pyrekordbox upgrade before distributing a new release.

flask>=3.0
flask-limiter>=3.5
werkzeug>=3.0
# Web framework — routes, SSE streaming, JSON responses, rate limiting.
# werkzeug is a Flask dependency but is imported directly in app.py.

waitress>=3.0
# Production WSGI server (Automator-safe: no stdout bleed unlike dev server)

numpy>=1.26
# Iron (BPM, key, beat grid) is built directly on numpy; librosa also needs it.

pyrekordbox==0.4.4
# Pin any pyrekordbox upgrade before distributing a new release.

mutagen==1.47
# Logged FALLBACK for tag I/O. Anvil (anvil/) is the primary reader/writer for
# BPM and key; mutagen reads general metadata in the scanner, renamer, organizer
# and de-duplicator, and steps in when Anvil declines a container.

librosa>=0.10.2
# Logged FALLBACK only. Iron is the primary BPM/key detector; librosa stands in
# when Iron returns nothing for a file, decodes non-soundfile formats for the
# waveform generator, and estimates beat 1 for the USB export grid. Dropping it
# to optional is tracked in docs/IRON_RESEARCH.md (first-beat swap not yet a
# clear win on real tracks).

numba<0.62
# Pinned below the version pip resolves by default (0.62.1 as of 2026-08) because that
# version SEGFAULTS -- not raises an exception, an unrecoverable process crash -- inside
# librosa.beat.beat_track() on every real file tested, under Python 3.13. Confirmed via
# `python -X faulthandler`: the crash is inside a numba-JIT-compiled gufunc
# (numba/np/ufunc/gufunc.py) called from librosa/beat.py's __last_beat. That is the
# librosa fallback path and the USB-export first-beat estimate, so an unpinned numba
# silently breaks them. 0.61.2 confirmed working (docs/IRON_RESEARCH.md SS14.2) --
# re-validate before raising this pin.

# essentia is NOT a dependency any more (removed in v1.2.0; see
# requirements_optional.txt for why it must not come back).

soundfile>=0.12.1
# Audio file I/O — sf.info() for source bit-depth detection and for verifying
# normalisation output

psutil>=5.9
# System info: disk partitions and usage for the drives panel in app.py

pyacoustid==1.3
# Acoustic fingerprinting wrapper; requires fpcalc binary (see above)

yt-dlp>=2026.7.4
# Audio download from Bandcamp, Beatport, Soundcloud, and direct URLs.
# Requires ffmpeg (already a system dep) for post-download format conversion.
# Floor bumped 2026-08-01 (pip-audit: prior floor resolved to a version with
# 8 known CVEs). yt-dlp is calendar-versioned with no stable API surface
# FableGear depends on, so no ceiling is pinned here.

mcp>=1.28.1,<2.0.0
# Anthropic Model Context Protocol SDK — used by mcp_server.py to expose
# FableGear tools to Claude Desktop, Cursor, Windsurf, and other MCP clients.
# Supports both stdio transport (local agents) and SSE transport (HTTP clients).
# Floor is 1.28.1 for the WebSocket origin-validation fix (PYSEC-2026-3483,
# only reachable via --transport sse). Ceiling excludes 2.0.0: it renames/
# restructures the import surface mcp_server.py depends on (confirmed by
# `import mcp_server` failing its own "mcp not installed" check against
# 2.0.0 despite the package being present) — re-pin only after porting
# mcp_server.py to the 2.x API.

pywebview>=5.0
# Native OS window (WKWebView on macOS) — replaces the browser launch step.
# Enables real drag-and-drop path delivery and a proper .app experience.

# pytest lives in requirements-dev.txt — it is a development tool and every
# end user was installing it (and its dependencies) for no reason.

# PyQt5 removed — splash_player.py was dead code, never invoked by launch.sh
# pyinstaller removed — build-only tool, see requirements_build.txt
